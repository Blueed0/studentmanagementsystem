UML Class Diagram

classDiagram
    direction TB
    class IIdentifiable {
        <>
        +string Id
    }
    class Student {
        <>
        -string _id
        -string _fullName
        -int _age
        -string _degreeProgram
        +string Id
        +string FullName
        +int Age
        +string DegreeProgram
        +GetAcademicLevel()* string
        +GetStudentDetails() string
    }
    class BachelorStudent {
        -int _studyYear
        +int StudyYear
        +GetAcademicLevel() string
        +GetStudentDetails() string
    }
    class MasterStudent {
        -string _thesisTopic
        +string ThesisTopic
        +GetAcademicLevel() string
        +GetStudentDetails() string
    }
    class Course {
        -string _id
        -string _title
        -int _capacity
        +string Id
        +string Title
        +int Capacity
        +GetCourseDetails() string
    }
    class Enrollment {
        -string _enrollmentId
        -Student _student
        -Course _course
        -DateTime _enrollmentDate
        +string EnrollmentId
        +Student Student
        +Course Course
        +DateTime EnrollmentDate
        +GetEnrollmentDetails() string
    }
    class UniversityManager {
        -List~Student~ _students
        -List~Course~ _courses
        -List~Enrollment~ _enrollments
        +RegisterBachelorStudent(...)
        +RegisterMasterStudent(...)
        +CreateCourse(...)
        +EnrollStudent(...)
        +DropEnrollment(...)
        +GetStudentsByProgram(...)
        +CountEnrollmentsForStudent(...)
    }
    class Program {
        -UniversityManager _manager$
        +Main(args)$ void
    }
    IIdentifiable <|.. Student

Main Menu Flowchart

flowchart TD
    A([Start Program]) --> B[Load sample seed data]
    B --> C[Show Main Menu]
    C --> D[Read menu option]
    D --> E{Option?}
    E -->|1-14 features| F[Run selected handler]
    E -->|0| T([Exit])
    E -->|invalid| U[Show error]
    F --> V[Pause / Enter]
    U --> V
    V --> C

Enroll Student Decision Flow

flowchart TD
    A([Enroll Student]) --> B[Read Student Id and Course Code]
    B --> C{Student found?}
    C -->|No| R1[Reject unknown student]
    C -->|Yes| D{Course found?}
    D -->|No| R2[Reject unknown course]
    D -->|Yes| E{Already enrolled?}
    E -->|Yes| R3[Reject duplicate]
    E -->|No| F{Seats remaining?}
    F -->|No| R4[Reject course full]
    F -->|Yes| G{Courses under max 5?}
    G -->|No| R5[Reject max courses]
    G -->|Yes| H[Create and store Enrollment]
    H --> S([Success])
    R1 --> X([State unchanged])
    R2 --> X
    R3 --> X
    R4 --> X
    R5 --> X
